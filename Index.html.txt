    <script>
        const WEB_APP_URL = "https://script.google.com/macros/s/AKfycbzU1yXWPzKL9y6yiNw1JBy2XkzKLUKlNCy4LSxcLNYHi8h-Iufn3gbE23giNoPUJW-2/exec";

        let currentPropertyMatches = [];
        let currentClientMatches = [];

        window.addEventListener("DOMContentLoaded", loadDropdownOptions);

        async function loadDropdownOptions() {
            const statusBadge = document.getElementById('statusBadge');
            const propSelect = document.getElementById('propertySelect');
            const clientSelect = document.getElementById('clientSelect');

            try {
                const response = await fetch(`${WEB_APP_URL}?action=getDropdownData`);
                const data = await response.json();

                if (data.properties && data.properties.length > 0) {
                    propSelect.innerHTML = `<option value="">-- Choose a Property Listing --</option>`;
                    data.properties.forEach(p => {
                        propSelect.innerHTML += `<option value="${p.id}">${p.label}</option>`;
                    });
                } else {
                    propSelect.innerHTML = `<option value="">No Properties Found</option>`;
                }

                if (data.clients && data.clients.length > 0) {
                    clientSelect.innerHTML = `<option value="">-- Choose a Client Lead --</option>`;
                    data.clients.forEach(c => {
                        clientSelect.innerHTML += `<option value="${c.id}">${c.label}</option>`;
                    });
                } else {
                    clientSelect.innerHTML = `<option value="">No Clients Found</option>`;
                }

                statusBadge.className = "bg-emerald-100 text-emerald-800 text-xs font-semibold px-3 py-1 rounded-full";
                statusBadge.textContent = "Live Sync Active";

            } catch (error) {
                console.error("Error loading dropdown data:", error);
                statusBadge.className = "bg-red-100 text-red-800 text-xs font-semibold px-3 py-1 rounded-full";
                statusBadge.textContent = "Connection Error";
            }
        }

        async function runMatchByProperty() {
            const propId = document.getElementById('propertySelect').value;
            const resultsContainer = document.getElementById('propertyMatchResults');
            const propControls = document.getElementById('propControls');
            
            if (!propId) {
                alert("Please select a property from the dropdown.");
                return;
            }

            resultsContainer.innerHTML = `<p class="text-sm text-slate-500 animate-pulse">Searching matching clients...</p>`;
            propControls.classList.add('hidden');

            try {
                const response = await fetch(`${WEB_APP_URL}?action=matchClients&id=${propId}`);
                currentPropertyMatches = await response.json();

                if (currentPropertyMatches.length === 0) {
                    resultsContainer.innerHTML = `<p class="text-sm text-red-500 bg-red-50 p-3 rounded-lg border border-red-200">No matching clients found for this property.</p>`;
                    return;
                }

                propControls.classList.remove('hidden');
                document.getElementById('propSearchFilter').value = '';
                renderPropertyResults(currentPropertyMatches);
            } catch (error) {
                resultsContainer.innerHTML = `<p class="text-sm text-red-500">Error fetching matching results.</p>`;
            }
        }

        function renderPropertyResults(items) {
            const container = document.getElementById('propertyMatchResults');
            if (items.length === 0) {
                container.innerHTML = `<p class="text-sm text-slate-400 italic">No matches fit the filter criteria.</p>`;
                return;
            }

            let html = `<p class="text-xs font-semibold text-emerald-600 mb-2">Showing ${items.length} Matching Client(s):</p>`;
            items.forEach(client => {
                html += `
                    <div class="p-3 bg-slate-50 rounded-lg border border-slate-200 text-sm hover:border-emerald-300 transition">
                        <p class="font-bold text-slate-800">${client.name} <span class="text-xs font-normal text-slate-500">(Client ID: ${client.clientID})</span></p>
                        <p class="text-xs text-slate-600">Phone: ${client.phone} | Location: ${client.location}</p>
                        <p class="text-xs font-medium text-emerald-700 mt-1">Intended Use: ${client.intendedUse}</p>
                    </div>`;
            });
            container.innerHTML = html;
        }

        function filterPropertyResults() {
            const query = document.getElementById('propSearchFilter').value.toLowerCase();
            const filtered = currentPropertyMatches.filter(item => 
                String(item.name).toLowerCase().includes(query) ||
                String(item.location).toLowerCase().includes(query) ||
                String(item.phone).toLowerCase().includes(query) ||
                String(item.intendedUse).toLowerCase().includes(query)
            );
            renderPropertyResults(filtered);
        }

        async function runMatchByClient() {
            const clientId = document.getElementById('clientSelect').value;
            const resultsContainer = document.getElementById('clientMatchResults');
            const clientControls = document.getElementById('clientControls');
            
            if (!clientId) {
                alert("Please select a client from the dropdown.");
                return;
            }

            resultsContainer.innerHTML = `<p class="text-sm text-slate-500 animate-pulse">Searching matching properties...</p>`;
            clientControls.classList.add('hidden');

            try {
                const response = await fetch(`${WEB_APP_URL}?action=matchProperties&id=${clientId}`);
                currentClientMatches = await response.json();

                if (currentClientMatches.length === 0) {
                    resultsContainer.innerHTML = `<p class="text-sm text-red-500 bg-red-50 p-3 rounded-lg border border-red-200">No matching properties found for this client.</p>`;
                    return;
                }

                clientControls.classList.remove('hidden');
                document.getElementById('clientSearchFilter').value = '';
                renderClientResults(currentClientMatches);
            } catch (error) {
                resultsContainer.innerHTML = `<p class="text-sm text-red-500">Error fetching matching results.</p>`;
            }
        }

        function renderClientResults(items) {
            const container = document.getElementById('clientMatchResults');
            if (items.length === 0) {
                container.innerHTML = `<p class="text-sm text-slate-400 italic">No matches fit the filter criteria.</p>`;
                return;
            }

            let html = `<p class="text-xs font-semibold text-emerald-600 mb-2">Showing ${items.length} Matching Property(ies):</p>`;
            items.forEach(prop => {
                html += `
                    <div class="p-3 bg-slate-50 rounded-lg border border-slate-200 text-sm hover:border-emerald-300 transition">
                        <p class="font-bold text-slate-800">Property ID: ${prop.propertyID} <span class="text-xs font-normal text-slate-500">(Key Man: ${prop.keyMan})</span></p>
                        <p class="text-xs text-slate-600">Location: ${prop.location} | Size: ${prop.size}</p>
                        <p class="text-xs font-semibold text-emerald-700 mt-1">Asking Price: ${prop.price}</p>
                    </div>`;
            });
            container.innerHTML = html;
        }

        function filterClientResults() {
            const query = document.getElementById('clientSearchFilter').value.toLowerCase();
            const filtered = currentClientMatches.filter(item => 
                String(item.propertyID).toLowerCase().includes(query) ||
                String(item.keyMan).toLowerCase().includes(query) ||
                String(item.location).toLowerCase().includes(query) ||
                String(item.price).toLowerCase().includes(query)
            );
            renderClientResults(filtered);
        }

        function exportToCSV(dataArray, filename) {
            if (!dataArray || dataArray.length === 0) {
                alert("No data available to export.");
                return;
            }

            const headers = Object.keys(dataArray[0]);
            const csvRows = [headers.join(",")];

            dataArray.forEach(row => {
                const values = headers.map(header => {
                    const val = String(row[header] || '').replace(/"/g, '""');
                    return `"${val}"`;
                });
                csvRows.push(values.join(","));
            });

            const csvContent = "data:text/csv;charset=utf-8," + csvRows.join("\n");
            const encodedUri = encodeURI(csvContent);
            const link = document.createElement("a");
            link.setAttribute("href", encodedUri);
            link.setAttribute("download", filename);
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
        }
