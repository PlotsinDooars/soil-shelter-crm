function verifyDatabaseSchema() {
  var ss = getDb();
  var expectedSheets = [
    { name: SHEET_CLIENTS, headers: ["Client ID","Date","Name","Phone","Address","Intended Use","Specialty","Area Category","Owner Category","Location","Min Size","Max Size","Size Unit","Min Budget","Max Budget","Budget Unit","Next Follow Up","Status","Remarks"] },

  ];

  for (var i = 0; i < expectedSheets.length; i++) {
    var item = expectedSheets[i];
    var sheet = ss.getSheetByName(item.name);
    if (!sheet) throw new Error("Missing tab in Google Sheet: '" + item.name + "'");
  }
  return true;
}

function getNextSequentialId(sheetName, idColumnIndex) {
  var lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    var sheet = getDb().getSheetByName(sheetName);
    var lastRow = sheet.getLastRow();
    if (lastRow <= 1) return 1;
    
    var ids = sheet.getRange(2, idColumnIndex, lastRow - 1, 1).getValues();
    var maxId = 0;
    for (var i = 0; i < ids.length; i++) {
      var val = parseInt(ids[i][0], 10);
      if (!isNaN(val) && val > maxId) maxId = val;
    }
    return maxId + 1;
  } finally {
    lock.releaseLock();
  }
}

function getDashboardData() {
  verifyDatabaseSchema();
  var ss = getDb();
  
  var clientSheet = ss.getSheetByName(SHEET_CLIENTS);
  var propSheet = ss.getSheetByName(SHEET_PROPERTIES);
  var taskSheet = ss.getSheetByName(SHEET_TASK);
  
  var clientData = clientSheet.getLastRow() > 1 ? clientSheet.getRange(2, 1, clientSheet.getLastRow() - 1, 19).getValues() : [];
  var propData = propSheet.getLastRow() > 1 ? propSheet.getRange(2, 1, propSheet.getLastRow() - 1, 25).getValues() : [];
  var taskData = taskSheet.getLastRow() > 1 ? taskSheet.getRange(2, 1, taskSheet.getLastRow() - 1, 8).getValues() : [];
  
  var now = new Date();
  now.setHours(0, 0, 0, 0);

  var totalClients = clientData.length;
  var totalProperties = propData.length;
  var totalTasks = taskData.length;
  
  var followupsDueToday = 0;
  var followupsOverdue = 0;
  var totalFollowupsScheduled = 0;

  for (var c = 0; c < clientData.length; c++) {
    var row = clientData[c];
    var nextFollowUp = row[16];
    var status = String(row[17]).trim().toLowerCase();
    
    if (status === 'converted' || status === 'close deal' || status === 'not interested') continue;

    if (nextFollowUp) {
      totalFollowupsScheduled++;
      var fDate = new Date(nextFollowUp);
      if (!isNaN(fDate.getTime())) {
        fDate.setHours(0,0,0,0);
        if (fDate.getTime() === now.getTime()) {
          followupsDueToday++;
        } else if (fDate.getTime() < now.getTime()) {
          followupsOverdue++;
        }
      }
    }
  }

  var todayStr = formatDateYMD(new Date());
  var todaysTasks = [];
  for (var t = 0; t < taskData.length; t++) {
    var tRow = taskData[t];
    var tDate = tRow[0];
    if (tDate) {
      var formattedTDate = formatDateYMD(new Date(tDate));
      if (formattedTDate === todayStr) {
        todaysTasks.push({
          rowIndex: t + 2,
          date: formattedTDate,
          task: String(tRow[1]),
          type: String(tRow[2]),
          place: String(tRow[3]),
          time: String(tRow[4]),
          who: String(tRow[5]),
          status: String(tRow[6]),
          remarks: String(tRow[7])
        });
      }
    }
  }

  return {
    totalClients: totalClients,
    totalProperties: totalProperties,
    totalTasks: totalTasks,
    followupsDue: totalFollowupsScheduled,
    followupsOverdue: followupsOverdue,
    followupsDueToday: followupsDueToday,
    todaysTasks: todaysTasks,
    lastSynced: new Date().toLocaleTimeString()
  };
}

function saveClientRecord(data) {
  var lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    verifyDatabaseSchema();
    var sheet = getDb().getSheetByName(SHEET_CLIENTS);
    
    var isEdit = data.clientId && parseInt(data.clientId, 10) > 0;
    var clientId = isEdit ? parseInt(data.clientId, 10) : getNextSequentialId(SHEET_CLIENTS, 1);

    var rowValues = [
      clientId,
      data.date || formatDateYMD(new Date()),
      data.name || '',
      data.phone || '',
      data.address || '',
      data.intendedUse || '',
      data.specialty || '',
      data.areaCategory || '',
      data.ownerCategory || '',
      data.location || '',
      data.minSize !== "" ? Number(data.minSize) : '',
      data.maxSize !== "" ? Number(data.maxSize) : '',
      data.sizeUnit || '',
      data.minBudget !== "" ? Number(data.minBudget) : '',
      data.maxBudget !== "" ? Number(data.maxBudget) : '',
      data.budgetUnit || '',
      data.nextFollowUp || '',
      data.status || 'New',
      data.remarks || ''
    ];

    if (isEdit) {
      var lastRow = sheet.getLastRow();
      var found = false;
      if (lastRow > 1) {
        var ids = sheet.getRange(2, 1, lastRow - 1, 1).getValues();
        for (var i = 0; i < ids.length; i++) {
          if (parseInt(ids[i][0], 10) === clientId) {
            sheet.getRange(i + 2, 1, 1, 19).setValues([rowValues]);
            found = true;
            break;
          }
        }
      }
      if (!found) throw new Error("Client ID #" + clientId + " not found.");
    } else {
      sheet.appendRow(rowValues);
    }
    return { success: true, clientId: clientId };
  } finally {
    lock.releaseLock();
  }
}

function matchMakerByProperty(propertyId) {
  var props = getAllProperties();
  var clients = getAllClients();

  var property = props.find(function(p) { return parseInt(p.propertyId, 10) === parseInt(propertyId, 10); });
  if (!property) return { property: null, matches: [] };

  var matches = [];

  clients.forEach(function(c) {
    if (c.status === 'Not Interested' || c.status === 'Converted' || c.status === 'Close Deal') return;

    var score = 0;
    var matchReasons = [];
    var mismatchReasons = [];

    if (c.intendedUse && property.intendedUse) {
      var cUses = c.intendedUse.split(',').map(function(s) { return s.trim().toLowerCase(); });
      var pUses = property.intendedUse.split(',').map(function(s) { return s.trim().toLowerCase(); });
      var useMatch = cUses.some(function(u) { return pUses.indexOf(u) !== -1; });
      if (useMatch) {
        score += 40;
        matchReasons.push("Intended Use Matches (" + c.intendedUse + ")");
      } else {
        mismatchReasons.push("Intended Use Mismatch");
      }
    } else {
      score += 20;
    }

    if (c.location && property.location) {
      var cLocs = c.location.split(',').map(function(s) { return s.trim().toLowerCase(); });
      var pLocs = property.location.split(',').map(function(s) { return s.trim().toLowerCase(); });
      var locMatch = cLocs.some(function(l) { return l === 'not specific' || pLocs.indexOf(l) !== -1; });
      if (locMatch) {
        score += 30;
        matchReasons.push("Location Matches (" + property.location + ")");
      } else {
        mismatchReasons.push("Location Mismatch");
      }
    } else {
      score += 15;
    }

    if (c.specialty && property.specialty) {
      var cSpecs = c.specialty.split(',').map(function(s) { return s.trim().toLowerCase(); });
      var pSpecs = property.specialty.split(',').map(function(s) { return s.trim().toLowerCase(); });
      var specMatch = cSpecs.some(function(sp) { return pSpecs.indexOf(sp) !== -1; });
      if (specMatch) {
        score += 30;
        matchReasons.push("Specialty Matches (" + property.specialty + ")");
      }
    } else {
      score += 15;
    }

    if (score > 30) {
      matches.push({
        client: c,
        matchScore: Math.min(100, score),
        matchReasons: matchReasons,
        mismatchReasons: mismatchReasons
      });
    }
  });

  matches.sort(function(a, b) { return b.matchScore - a.matchScore; });
  return { property: property, matches: matches };
}

function formatDateYMD(d) {
  if (!d) return '';
  var date = new Date(d);
  if (isNaN(date.getTime())) return String(d);
  var yyyy = date.getFullYear();
  var mm = String(date.getMonth() + 1).padStart(2, '0');
  var dd = String(date.getDate()).padStart(2, '0');
  return yyyy + '-' + mm + '-' + dd;
}
