function buildChart(containerId, optionsArray, groupName, otherInputId) {
  var container = document.getElementById(containerId);
  if (!container) return;
  container.innerHTML = '';

  optionsArray.forEach(function(opt) {
    var label = document.createElement('label');
    label.className = 'option-checkbox-item';
    
    var cb = document.createElement('input');
    cb.type = 'checkbox';
    cb.name = groupName;
    cb.value = opt;

    if (opt === 'OTHERS' && otherInputId) {
      cb.onchange = function() {
        var oth = document.getElementById(otherInputId);
        if (cb.checked) oth.classList.remove('hidden');
        else oth.classList.add('hidden');
      };
    }

    label.appendChild(cb);
    label.appendChild(document.createTextNode(opt));
    container.appendChild(label);
  });
}

function getChartSelectedValues(groupName, otherInputId) {
  var checkboxes = document.querySelectorAll('input[name="' + groupName + '"]:checked');
  var results = [];
  checkboxes.forEach(function(cb) {
    var val = cb.value;
    if (val === 'OTHERS' && otherInputId) {
      var othVal = document.getElementById(otherInputId).value.trim();
      if (othVal) results.push("OTHERS: " + othVal);
      else results.push("OTHERS");
    } else {
      results.push(val);
    }
  });
  return results.join(', ');
}

function setChartSelectedValues(groupName, otherInputId, rawString) {
  var checkboxes = document.querySelectorAll('input[name="' + groupName + '"]');
  checkboxes.forEach(function(cb) { cb.checked = false; });
  if (otherInputId) document.getElementById(otherInputId).classList.add('hidden');

  if (!rawString) return;
  var items = String(rawString).split(',').map(function(s) { return s.trim(); });

  items.forEach(function(item) {
    if (item.indexOf('OTHERS:') === 0 && otherInputId) {
      checkboxes.forEach(function(cb) { if (cb.value === 'OTHERS') cb.checked = true; });
      var oth = document.getElementById(otherInputId);
      oth.value = item.replace('OTHERS:', '').trim();
      oth.classList.remove('hidden');
    } else {
      checkboxes.forEach(function(cb) { if (cb.value === item) cb.checked = true; });
    }
  });
}

function renderFollowups() {
  var overdueDiv = document.getElementById('followup-list-overdue');
  var todayDiv = document.getElementById('followup-list-today');
  var upcomingDiv = document.getElementById('followup-list-upcoming');

  overdueDiv.innerHTML = ''; todayDiv.innerHTML = ''; upcomingDiv.innerHTML = '';
  var now = new Date(); now.setHours(0,0,0,0);

  state.clients.forEach(function(c) {
    if (c.status === 'Converted' || c.status === 'Close Deal' || !c.rawNextFollowUp) return;
    var fDate = new Date(c.rawNextFollowUp);
    if (isNaN(fDate.getTime())) return;
    fDate.setHours(0,0,0,0);

    var card = document.createElement('div');
    card.className = 'card card-blue';
    card.style.marginBottom = '8px';
    card.innerHTML = 
      '<div style="flex:1">' +
        '<strong>#' + c.clientId + ' - ' + escapeHtml(c.name) + '</strong> (' + c.phone + ')<br>' +
        '<small>Follow Up: ' + c.nextFollowUp + ' | Intended: ' + escapeHtml(c.intendedUse) + '</small>' +
      '</div>' +
      '<button class="btn-primary" onclick="openEditClientForm(' + c.clientId + ')">Update</button>';

    if (fDate.getTime() < now.getTime()) overdueDiv.appendChild(card);
    else if (fDate.getTime() === now.getTime()) todayDiv.appendChild(card);
    else upcomingDiv.appendChild(card);
  });

  if (!overdueDiv.hasChildNodes()) overdueDiv.innerHTML = '<p>No overdue follow-ups!</p>';
  if (!todayDiv.hasChildNodes()) todayDiv.innerHTML = '<p>No follow-ups due today.</p>';
  if (!upcomingDiv.hasChildNodes()) upcomingDiv.innerHTML = '<p>No upcoming follow-ups scheduled.</p>';
}

  if (mode === 'property') {
    document.getElementById('matchmaker-title').innerText = 'MATCHMAKER BY PROPERTY';
    document.getElementById('matchmaker-select-label').innerText = 'Select Property to Find Matching Clients:';
    document.getElementById('matchmaker-options').classList.add('hidden');
    state.properties.forEach(function(p) {
      var opt = document.createElement('option');
      opt.value = p.propertyId;
      opt.innerText = 'Property #' + p.propertyId + ' - ' + p.location + ' (' + p.price + ' ' + p.priceUnit + ')';
      sel.appendChild(opt);
    });
  } else {
    document.getElementById('matchmaker-title').innerText = 'MATCHMAKER BY CLIENT';
    document.getElementById('matchmaker-select-label').innerText = 'Select Client Requirement to Find Properties:';
    document.getElementById('matchmaker-options').classList.remove('hidden');
    state.clients.forEach(function(c) {
      var opt = document.createElement('option');
      opt.value = c.clientId;
      opt.innerText = 'Client #' + c.clientId + ' - ' + c.name + ' (' + c.intendedUse + ')';
      sel.appendChild(opt);
    });
  }

  document.getElementById('matchmaker-results-container').innerHTML = '<div class="placeholder-text">Select a record above to run the matching engine.</div>';
  document.getElementById('modal-matchmaker').classList.remove('hidden');
}

function executeMatchmaking() {
  var id = document.getElementById('matchmaker-select-item').value;
  if (!id) return;
  showLoading('Running Matchmaking Engine...');

  if (state.currentMatchmakerMode === 'property') {
    google.script.run.withSuccessHandler(renderPropertyMatches).matchMakerByProperty(id);
  } else {
    var includeSold = document.getElementById('match-include-sold').checked;
    google.script.run.withSuccessHandler(renderClientMatches).matchMakerByClient(id, includeSold);
  }
}

function renderClientMatches(res) {
  hideLoading();
  var container = document.getElementById('matchmaker-results-container');
  container.innerHTML = '<h4>Matches for Client #' + res.client.clientId + ' (' + escapeHtml(res.client.name) + ')</h4>';
  if (!res.matches || res.matches.length === 0) {
    container.innerHTML += '<p>No matching properties found.</p>'; return;
  }
  res.matches.forEach(function(m) {
    var box = document.createElement('div');
    box.style.border = '1px solid #ccc'; box.style.borderRadius = '8px'; box.style.padding = '10px'; box.style.marginBottom = '8px';
    box.innerHTML = 
      '<div style="display:flex; justify-content:space-between; font-weight:bold;">' +
        '<span>Property #' + m.property.propertyId + ' - ' + escapeHtml(m.property.location) + '</span>' +
        '<span style="color:#2e7d32">' + m.matchScore + '% Match</span>' +
      '</div>' +
      '<div>Price: ' + m.property.price + ' ' + m.property.priceUnit + ' | Size: ' + m.property.size + ' ' + m.property.sizeUnit + '</div>' +
      '<div style="font-size:12px; color:green; margin-top:4px;">✔ ' + m.matchReasons.join('<br>✔ ') + '</div>' +
      (m.mismatchReasons.length > 0 ? '<div style="font-size:12px; color:red; margin-top:4px;">✖ ' + m.mismatchReasons.join('<br>✖ ') + '</div>' : '');
    container.appendChild(box);
  });
}

function filterClientsTable() {
  var q = document.getElementById('client-search').value.toLowerCase();
  var st = document.getElementById('client-filter-status').value;
  document.querySelectorAll('#clients-tbody tr').forEach(function(row) {
    var text = row.innerText.toLowerCase();
    row.style.display = ((!q || text.indexOf(q) !== -1) && (!st || text.indexOf(st.toLowerCase()) !== -1)) ? '' : 'none';
  });
}
