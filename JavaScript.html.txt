<script>
/**
 * SOIL & SHELTER CRM — Frontend Controller Logic
 * Manages UI interactions, real-time polling, form lifecycle, WhatsApp/Call shortcuts,
 * and server calls via google.script.run.
 */

var state = {
  clients: [],
  properties: [],
  tasks: [],
  dashboard: {},
  currentMatchmakerMode: 'property'
};

// Initialize Application
document.addEventListener('DOMContentLoaded', function() {
  fetchUserInfo();
  refreshAllData();
  
  // Setup 10-second polling sync
  setInterval(function() {
    refreshAllData(true); // Silent background refresh
  }, 10000);

  // Setup Other choices listener
  setupOtherInputTriggers();
});

function showLoading(text) {
  document.getElementById('loading-text').innerText = text || 'Loading...';
  document.getElementById('loading-overlay').classList.remove('hidden');
}

function hideLoading() {
  document.getElementById('loading-overlay').classList.add('hidden');
}

function showToast(msg) {
  var t = document.getElementById('toast-message');
  t.innerText = msg;
  t.classList.remove('hidden');
  setTimeout(function() { t.classList.add('hidden'); }, 3500);
}

function fetchUserInfo() {
  google.script.run
    .withSuccessHandler(function(info) {
      document.getElementById('user-email-display').innerText = info.email;
    })
    .getCurrentUserInfo();
}

function refreshAllData(silent) {
  if (!silent) showLoading('Syncing with Google Sheets...');
  
  google.script.run
    .withSuccessHandler(function(dashData) {
      state.dashboard = dashData;
      renderDashboard();
      document.getElementById('last-synced-time').innerText = dashData.lastSynced;
      
      // Fetch Detailed Lists
      fetchListsData(silent);
    })
    .withFailureHandler(function(err) {
      hideLoading();
      showToast('Sync Error: ' + err.message);
    })
    .getDashboardData();
}

function fetchListsData(silent) {
  google.script.run.withSuccessHandler(function(clients) {
    state.clients = clients;
    renderClientsTable();
    renderFollowups();
    
    google.script.run.withSuccessHandler(function(props) {
      state.properties = props;
      renderPropertiesTable();
      
      google.script.run.withSuccessHandler(function(tasks) {
        state.tasks = tasks;
        renderTasksTable();
        hideLoading();
      }).getAllTasks();
    }).getAllProperties();
  }).getAllClients();
}

function refreshCurrentView() {
  refreshAllData(false);
}

// Navigation
function navigateTo(viewId) {
  document.querySelectorAll('.app-view').forEach(function(v) { v.classList.remove('active'); });
  document.querySelectorAll('.nav-item').forEach(function(n) { n.classList.remove('active'); });
  
  document.getElementById(viewId).classList.add('active');
  
  // Highlight nav bar button
  if (viewId === 'view-dashboard') document.querySelectorAll('.nav-item')[0].classList.add('active');
  if (viewId === 'view-clients') document.querySelectorAll('.nav-item')[1].classList.add('active');
  if (viewId === 'view-properties') document.querySelectorAll('.nav-item')[2].classList.add('active');
  if (viewId === 'view-tasks') document.querySelectorAll('.nav-item')[3].classList.add('active');
  if (viewId === 'view-followups') document.querySelectorAll('.nav-item')[4].classList.add('active');
}

// Render Dashboard
function renderDashboard() {
  var d = state.dashboard;
  document.getElementById('dash-total-clients').innerText = d.totalClients || 0;
  document.getElementById('dash-total-properties').innerText = d.totalProperties || 0;
  document.getElementById('dash-total-tasks').innerText = d.totalTasks || 0;
  document.getElementById('dash-followups-due').innerText = d.followupsDue || 0;
  document.getElementById('dash-overdue-subtext').innerHTML = '<i class="fa-solid fa-triangle-exclamation"></i> ' + (d.followupsOverdue || 0) + ' Overdue';

  // Todays Task List
  var ol = document.getElementById('todays-task-ol');
  ol.innerHTML = '';
  if (d.todaysTasks && d.todaysTasks.length > 0) {
    d.todaysTasks.forEach(function(t) {
      var li = document.createElement('li');
      li.innerText = t.time + ' - ' + t.task + ' (' + t.who + ') [' + t.status + ']';
      li.onclick = function() { openEditTaskForm(t.rowIndex); };
      ol.appendChild(li);
    });
  } else {
    ol.innerHTML = '<li>No tasks scheduled for today.</li>';
  }
}

// FOLLOW-UPS RENDERING
// ==========================================
function renderFollowups() {
  var overdueDiv = document.getElementById('followup-list-overdue');
  var todayDiv = document.getElementById('followup-list-today');
  var upcomingDiv = document.getElementById('followup-list-upcoming');

  overdueDiv.innerHTML = '';
  todayDiv.innerHTML = '';
  upcomingDiv.innerHTML = '';

  var now = new Date();
  now.setHours(0,0,0,0);

  state.clients.forEach(function(c) {
    if (c.status === 'Converted' || c.status === 'Close Deal' || !c.rawNextFollowUp) return;

    var fDate = new Date(c.rawNextFollowUp);
    if (isNaN(fDate.getTime())) return;
    fDate.setHours(0,0,0,0);

    var card = document.createElement('div');
    card.className = 'card card-blue';
    card.style.marginBottom = '8px';
    card.innerHTML = 
      '<div style="flex:1">' +
        '<strong>#' + c.clientId + ' - ' + escapeHtml(c.name) + '</strong> (' + c.phone + ')<br>' +
        '<small>Follow Up: ' + c.nextFollowUp + ' | Intended: ' + escapeHtml(c.intendedUse) + '</small>' +
      '</div>' +
      '<button class="btn-primary" onclick="openEditClientForm(' + c.clientId + ')">Update</button>';

    if (fDate.getTime() < now.getTime()) {
      overdueDiv.appendChild(card);
    } else if (fDate.getTime() === now.getTime()) {
      todayDiv.appendChild(card);
    } else {
      upcomingDiv.appendChild(card);
    }
  });

  if (!overdueDiv.hasChildNodes()) overdueDiv.innerHTML = '<p class="text-muted">No overdue follow-ups!</p>';
  if (!todayDiv.hasChildNodes()) todayDiv.innerHTML = '<p class="text-muted">No follow-ups due today.</p>';
  if (!upcomingDiv.hasChildNodes()) upcomingDiv.innerHTML = '<p class="text-muted">No upcoming follow-ups scheduled.</p>';
}

// ==========================================
function openMatchmakerModal(mode) {
  state.currentMatchmakerMode = mode;
  var sel = document.getElementById('matchmaker-select-item');
  sel.innerHTML = '<option value="">-- Choose Record --</option>';

  if (mode === 'property') {
    document.getElementById('matchmaker-title').innerText = 'MATCHMAKER BY PROPERTY';
    document.getElementById('matchmaker-select-label').innerText = 'Select Property to Find Matching Clients:';
    document.getElementById('matchmaker-options').classList.add('hidden');

    state.properties.forEach(function(p) {
      var opt = document.createElement('option');
      opt.value = p.propertyId;
      opt.innerText = 'Property #' + p.propertyId + ' - ' + p.location + ' (' + p.price + ' ' + p.priceUnit + ')';
      sel.appendChild(opt);
    });
  } else {
    document.getElementById('matchmaker-title').innerText = 'MATCHMAKER BY CLIENT';
    document.getElementById('matchmaker-select-label').innerText = 'Select Client Requirement to Find Properties:';
    document.getElementById('matchmaker-options').classList.remove('hidden');

    state.clients.forEach(function(c) {
      var opt = document.createElement('option');
      opt.value = c.clientId;
      opt.innerText = 'Client #' + c.clientId + ' - ' + c.name + ' (' + c.intendedUse + ')';
      sel.appendChild(opt);
    });
  }

  document.getElementById('matchmaker-results-container').innerHTML = '<div class="placeholder-text">Select a record above to run the matching engine.</div>';
  document.getElementById('modal-matchmaker').classList.remove('hidden');
}

function executeMatchmaking() {
  var id = document.getElementById('matchmaker-select-item').value;
  if (!id) return;

  showLoading('Running Matchmaking Engine...');

  if (state.currentMatchmakerMode === 'property') {
    google.script.run
      .withSuccessHandler(renderPropertyMatches)
      .matchMakerByProperty(id);
  } else {
    var includeSold = document.getElementById('match-include-sold').checked;
    google.script.run
      .withSuccessHandler(renderClientMatches)
      .matchMakerByClient(id, includeSold);
  }
}

function renderClientMatches(res) {
  hideLoading();
  var container = document.getElementById('matchmaker-results-container');
  container.innerHTML = '<h4>Matches for Client #' + res.client.clientId + ' (' + escapeHtml(res.client.name) + ')</h4>';

  if (!res.matches || res.matches.length === 0) {
    container.innerHTML += '<p>No matching properties found based on current requirements.</p>';
    return;
  }

  res.matches.forEach(function(m) {
    var box = document.createElement('div');
    box.className = 'match-result-card';
    box.style.border = '1px solid #ccc';
    box.style.borderRadius = '8px';
    box.style.padding = '10px';
    box.style.marginBottom = '8px';
    box.style.background = '#fdfdfd';

    box.innerHTML = 
      '<div style="display:flex; justify-content:space-between; font-weight:bold;">' +
        '<span>Property #' + m.property.propertyId + ' - ' + escapeHtml(m.property.location) + '</span>' +
        '<span style="color:#2e7d32">' + m.matchScore + '% Match</span>' +
      '</div>' +
      '<div>Price: ' + m.property.price + ' ' + m.property.priceUnit + ' | Size: ' + m.property.size + ' ' + m.property.sizeUnit + '</div>' +
      '<div style="font-size:12px; color:green; margin-top:4px;">✔ ' + m.matchReasons.join('<br>✔ ') + '</div>' +
      (m.mismatchReasons.length > 0 ? '<div style="font-size:12px; color:red; margin-top:4px;">✖ ' + m.mismatchReasons.join('<br>✖ ') + '</div>' : '');

    container.appendChild(box);
  });
}

function renderPropertyMatches(res) {
  hideLoading();
  var container = document.getElementById('matchmaker-results-container');
  container.innerHTML = '<h4>Matching Clients for Property #' + res.property.propertyId + ' (' + escapeHtml(res.property.location) + ')</h4>';

  if (!res.matches || res.matches.length === 0) {
    container.innerHTML += '<p>No matching client requirements found.</p>';
    return;
  }

  res.matches.forEach(function(m) {
    var box = document.createElement('div');
    box.className = 'match-result-card';
    box.style.border = '1px solid #ccc';
    box.style.borderRadius = '8px';
    box.style.padding = '10px';
    box.style.marginBottom = '8px';
    box.style.background = '#fdfdfd';

    box.innerHTML = 
      '<div style="display:flex; justify-content:space-between; font-weight:bold;">' +
        '<span>Client #' + m.client.clientId + ' - ' + escapeHtml(m.client.name) + ' (' + m.client.phone + ')</span>' +
        '<span style="color:#2e7d32">' + m.matchScore + '% Match</span>' +
      '</div>' +
      '<div>Budget: ' + m.client.minBudget + '-' + m.client.maxBudget + ' ' + m.client.budgetUnit + '</div>' +
      '<div style="font-size:12px; color:green; margin-top:4px;">✔ ' + m.matchReasons.join('<br>✔ ') + '</div>' +
      (m.mismatchReasons.length > 0 ? '<div style="font-size:12px; color:red; margin-top:4px;">✖ ' + m.mismatchReasons.join('<br>✖ ') + '</div>' : '');

    container.appendChild(box);
  });
}

function getMultiSelectValues(selectId, otherInputId) {
  var select = document.getElementById(selectId);
  var result = [];
  for (var i = 0; i < select.options.length; i++) {
    if (select.options[i].selected) {
      var val = select.options[i].value;
      if (val === 'OTHERS' && otherInputId) {
        var otherVal = document.getElementById(otherInputId).value.trim();
        if (otherVal) result.push("OTHERS: " + otherVal);
      } else {
        result.push(val);
      }
    }
  }
  return result.join(', ');
}

function setMultiSelectValues(selectId, otherInputId, rawString) {
  var select = document.getElementById(selectId);
  if (!rawString) return;
  var items = String(rawString).split(',').map(function(s) { return s.trim(); });
  
  for (var i = 0; i < select.options.length; i++) {
    select.options[i].selected = false;
  }

  items.forEach(function(item) {
    if (item.indexOf('OTHERS:') === 0 && otherInputId) {
      for (var i = 0; i < select.options.length; i++) {
        if (select.options[i].value === 'OTHERS') select.options[i].selected = true;
      }
      document.getElementById(otherInputId).value = item.replace('OTHERS:', '').trim();
      document.getElementById(otherInputId).classList.remove('hidden');
    } else {
      for (var k = 0; k < select.options.length; k++) {
        if (select.options[k].value === item) select.options[k].selected = true;
      }
    }
  });
}

function getSelectWithOther(selectId, otherInputId) {
  var val = document.getElementById(selectId).value;
  if (val === 'Others' && otherInputId) {
    var oth = document.getElementById(otherInputId).value.trim();
    if (oth) return "Others: " + oth;
  }
  return val;
}

function setSelectWithOther(selectId, otherInputId, value) {
  if (!value) return;
  if (value.indexOf('Others:') === 0 && otherInputId) {
    document.getElementById(selectId).value = 'Others';
    document.getElementById(otherInputId).value = value.replace('Others:', '').trim();
    document.getElementById(otherInputId).classList.remove('hidden');
  } else {
    document.getElementById(selectId).value = value;
  }
}

function setupOtherInputTriggers() {
  ['c-intended', 'c-specialty', 'c-location', 'p-intended', 'p-specialty', 'p-location'].forEach(function(id) {
    var el = document.getElementById(id);
    if (el) {
      el.addEventListener('change', function() {
        var otherInput = document.getElementById(id + '-other');
        var hasOther = Array.from(el.selectedOptions).some(function(opt) { return opt.value === 'OTHERS'; });
        if (hasOther && otherInput) otherInput.classList.remove('hidden');
        else if (otherInput) otherInput.classList.add('hidden');
      });
    }
  });
}

function escapeHtml(str) {
  if (!str) return '';
  return String(str).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;');
}

function cleanPhoneForWa(phone) {
  if (!phone) return '';
  return String(phone).replace(/[^0-9]/g, '');
}

function formatDateForInput(dStr) {
  if (!dStr) return '';
  var d = new Date(dStr);
  if (isNaN(d.getTime())) return '';
  return d.toISOString().split('T')[0];
}

function filterClientsTable() {
  var q = document.getElementById('client-search').value.toLowerCase();
  var st = document.getElementById('client-filter-status').value;
  var rows = document.querySelectorAll('#clients-tbody tr');

  rows.forEach(function(row) {
    var text = row.innerText.toLowerCase();
    var matchSearch = !q || text.indexOf(q) !== -1;
    var matchStatus = !st || text.indexOf(st.toLowerCase()) !== -1;
    row.style.display = (matchSearch && matchStatus) ? '' : 'none';
  });
}
